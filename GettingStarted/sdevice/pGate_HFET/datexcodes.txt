"Data codes for semiconductor process and device simulation"


Variables {
    
MagnesiumActiveConcentration, MgActive {
    label     = "substitutional Magnesium concentration"
    symbol    = "MgActive"
    unit      = "cm^-3"
    factor    = 1.0e+12
    precision = 4
    interpol  = log
    material  = Semiconductor
}

MagnesiumConcentration, MagnesiumChemicalConcentration {
    label     = "total (chemical) Magnesium concentration"
    symbol    = "MgTotal"
    unit      = "cm^-3"
    factor    = 1.0e+12
    precision = 4
    interpol  = log
    material  = All
    property("floops") = "MgActive"
    doping    = acceptor (
                     active = MagnesiumActiveConcentration
                     ionized = MagnesiumMinusConcentration
                     material = AlN, GaN, AlGaN, InGaAs, InGaAsP, InP
                 )
}

MagnesiumMinusConcentration {

}
